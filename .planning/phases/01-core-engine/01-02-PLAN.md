Create the Objective-C audio engine following Apple's Hello Metronome pattern.

AudioEngine.h - Public interface:
```objc
#import <Foundation/Foundation.h>
#import <AVFoundation/AVFoundation.h>

NS_ASSUME_NONNULL_BEGIN

@interface AudioEngine : NSObject

@property (nonatomic, readonly) BOOL isPlaying;
@property (nonatomic) double bpm;  // 30-300
@property (nonatomic) NSInteger beatsPerBar;  // For accent on beat 1

- (instancetype)init;
- (BOOL)loadClickSoundFromURL:(NSURL *)clickURL accentURL:(NSURL *)accentURL error:(NSError **)error;
- (BOOL)start:(NSError **)error;
- (void)stop;

// For synchronization with visual/haptic
- (void)setOnBeatCallback:(void (^)(NSInteger beatNumber, BOOL isAccent))callback;

@end

NS_ASSUME_NONNULL_END
```

AudioEngine.m - Implementation (CRITICAL - follow these rules):
1. NO Swift calls from audio thread
2. NO locks in audio path
3. NO memory allocation in audio path
4. Pre-allocate ALL buffers before starting
5. Use timer-based scheduling, not just completion handlers

Key implementation details:

Properties:
- `AVAudioEngine *engine`
- `AVAudioPlayerNode *playerNode`
- `AVAudioPCMBuffer *clickBuffer` (pre-loaded)
- `AVAudioPCMBuffer *accentBuffer` (pre-loaded)
- `NSInteger beatInterval` (in samples)
- `NSInteger scheduledBeats` (how many beats scheduled ahead)
- `AVAudioFramePosition nextBeatSampleTime` (next beat's sample position)
- `NSInteger currentBeat` (0 to beatsPerBar-1)
- `dispatch_source_t schedulerTimer` (for proactive scheduling)

init:
- Create AVAudioEngine
- Create AVAudioPlayerNode
- Attach playerNode to engine
- Connect playerNode to mainMixerNode
- DO NOT start engine yet

loadClickSoundFromURL:
- Load WAV files into AVAudioPCMBuffer
- Store in clickBuffer and accentBuffer
- Return NO if files can't be loaded

start:
- Prepare and start engine
- Schedule first 3 beats immediately
- Start scheduler timer (fires 2x per beat interval)
- Timer callback schedules next beat to maintain 2-3 beat lookahead

Scheduler timer callback (CRITICAL PATTERN):
```objc
- (void)scheduleNextBeats {
    // Calculate current render time
    AVAudioTime *lastRenderTime = self.playerNode.lastRenderTime;
    if (!lastRenderTime) return;

    AVAudioFramePosition currentSampleTime = lastRenderTime.sampleTime;

    // Keep scheduling until we have 2-3 beats in the buffer
    while (self.nextBeatSampleTime < currentSampleTime + (self.beatInterval * 3)) {
        BOOL isAccent = (self.currentBeat == 0);
        AVAudioPCMBuffer *buffer = isAccent ? self.accentBuffer : self.clickBuffer;

        AVAudioTime *beatTime = [[AVAudioTime alloc] initWithSampleTime:self.nextBeatSampleTime
                                                               atRate:self.engine.mainMixerNode.outputFormat.sampleRate];

        [self.playerNode scheduleBuffer:buffer atTime:beatTime options:0 completionHandler:nil];

        // Fire callback on main thread (NOT in audio path)
        NSInteger beatNum = self.currentBeat;
        dispatch_async(dispatch_get_main_queue(), ^{
            if (self.onBeatCallback) {
                self.onBeatCallback(beatNum, isAccent);
            }
        });

        self.nextBeatSampleTime += self.beatInterval;
        self.currentBeat = (self.currentBeat + 1) % self.beatsPerBar;
    }
}
```

setBpm:
- Calculate new beatInterval in samples: (60.0 / bpm) * sampleRate
- Update beatInterval (will affect future scheduled beats)

stop:
- Stop playerNode
- Stop engine
- Cancel scheduler timer
- Reset state (nextBeatSampleTime, scheduledBeats, currentBeat)

scripts/generate-click-sounds.swift:
```swift
#!/usr/bin/env swift

import Foundation
import AVFoundation

// Generate a simple click sound (sine wave burst with envelope)
func generateClick(frequency: Double, duration: Double, attack: Double, decay: Double, amplitude: Float) -> [Float] {
    let sampleRate = 44100.0
    let numSamples = Int(duration * sampleRate)
    var samples = [Float](repeating: 0, count: numSamples)

    for i in 0..<numSamples {
        let t = Double(i) / sampleRate
        let envelope: Float

        if t < attack {
            envelope = Float(t / attack)
        } else if t < attack + decay {
            envelope = Float(1.0 - (t - attack) / decay)
        } else {
            envelope = 0
        }

        let wave = sin(2.0 * .pi * frequency * t)
        samples[i] = Float(wave) * envelope * amplitude
    }

    return samples
}

func writeWAV(samples: [Float], to url: URL) throws {
    let format = AVAudioFormat(standardFormatWithSampleRate: 44100, channels: 1)!
    let buffer = AVAudioPCMBuffer(pcmFormat: format, frameCapacity: AVAudioFrameCount(samples.count))!
    buffer.frameLength = AVAudioFrameCount(samples.count)

    let channelData = buffer.floatChannelData![0]
    for i in 0..<samples.count {
        channelData[i] = samples[i]
    }

    let file = try AVAudioFile(forWriting: url, settings: format.settings)
    try file.write(from: buffer)
}

// Generate click (800 Hz, 50ms)
let click = generateClick(frequency: 800, duration: 0.05, attack: 0.001, decay: 0.049, amplitude: 0.8)

// Generate accent (1200 Hz, 50ms, louder)
let accent = generateClick(frequency: 1200, duration: 0.05, attack: 0.001, decay: 0.049, amplitude: 1.0)

let resourcesDir = URL(fileURLWithPath: "Metronome/Resources")
try FileManager.default.createDirectory(at: resourcesDir, withIntermediateDirectories: true)

try writeWAV(samples: click, to: resourcesDir.appendingPathComponent("click.wav"))
try writeWAV(samples: accent, to: resourcesDir.appendingPathComponent("accent.wav"))

print("Generated click.wav and accent.wav in Metronome/Resources/")
```
