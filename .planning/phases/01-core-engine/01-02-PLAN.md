```swift
#!/usr/bin/env swift

import Foundation
import AVFoundation

// Generate a simple click sound (sine wave burst with envelope)
func generateClick(frequency: Double, duration: Double, attack: Double, decay: Double, amplitude: Float) -> [Float] {
    let sampleRate = 44100.0
    let numSamples = Int(duration * sampleRate)
    var samples = [Float](repeating: 0, count: numSamples)

    for i in 0..<numSamples {
        let t = Double(i) / sampleRate
        let envelope: Float

        if t < attack {
            envelope = Float(t / attack)
        } else if t < attack + decay {
            envelope = Float(1.0 - (t - attack) / decay)
        } else {
            envelope = 0
        }

        let wave = sin(2.0 * .pi * frequency * t)
        samples[i] = Float(wave) * envelope * amplitude
    }

    return samples
}

func writeWAV(samples: [Float], to url: URL) throws {
    let format = AVAudioFormat(standardFormatWithSampleRate: 44100, channels: 1)!
    let buffer = AVAudioPCMBuffer(pcmFormat: format, frameCapacity: AVAudioFrameCount(samples.count))!
    buffer.frameLength = AVAudioFrameCount(samples.count)

    let channelData = buffer.floatChannelData![0]
    for i in 0..<samples.count {
        channelData[i] = samples[i]
    }

    let file = try AVAudioFile(forWriting: url, settings: format.settings)
    try file.write(from: buffer)
}

// Generate click (800 Hz, 50ms)
let click = generateClick(frequency: 800, duration: 0.05, attack: 0.001, decay: 0.049, amplitude: 0.8)

// Generate accent (1200 Hz, 50ms, louder)
let accent = generateClick(frequency: 1200, duration: 0.05, attack: 0.001, decay: 0.049, amplitude: 1.0)

let resourcesDir = URL(fileURLWithPath: "Metronome/Resources")
try FileManager.default.createDirectory(at: resourcesDir, withIntermediateDirectories: true)

try writeWAV(samples: click, to: resourcesDir.appendingPathComponent("click.wav"))
try writeWAV(samples: accent, to: resourcesDir.appendingPathComponent("accent.wav"))

print("Generated click.wav and accent.wav in Metronome/Resources/")
```

```objc
#import "AudioEngine.h"

@interface AudioEngine ()

@property (nonatomic, strong) AVAudioEngine *engine;
@property (nonatomic, strong) AVAudioPlayerNode *playerNode;
@property (nonatomic, strong) AVAudioPCMBuffer *clickBuffer;
@property (nonatomic, strong) AVAudioPCMBuffer *accentBuffer;
@property (nonatomic, readwrite) BOOL isPlaying;
@property (nonatomic, copy) void (^onBeatCallback)(NSInteger beatNumber, BOOL isAccent);

// Scheduling state (will be used in Task 3)
@property (nonatomic) NSInteger beatInterval;  // in samples
@property (nonatomic) AVAudioFramePosition nextBeatSampleTime;
@property (nonatomic) NSInteger currentBeat;
@property (nonatomic, strong) dispatch_source_t schedulerTimer;

@end

@implementation AudioEngine

- (instancetype)init {
    self = [super init];
    if (self) {
        _bpm = 120.0;
        _beatsPerBar = 4;
        _isPlaying = NO;

        _engine = [[AVAudioEngine alloc] init];
        _playerNode = [[AVAudioPlayerNode alloc] init];

        [_engine attachNode:_playerNode];
    }
    return self;
}

- (void)dealloc {
    [self stop];
}

- (BOOL)loadClickSoundFromURL:(NSURL *)clickURL accentURL:(NSURL *)accentURL error:(NSError **)error {
    NSError *loadError = nil;

    // Load click sound
    AVAudioFile *clickFile = [[AVAudioFile alloc] initForReading:clickURL error:&loadError];
    if (!clickFile) {
        if (error) *error = loadError;
        return NO;
    }

    AVAudioFormat *format = clickFile.processingFormat;
    AVAudioFrameCount clickFrameCount = (AVAudioFrameCount)clickFile.length;
    self.clickBuffer = [[AVAudioPCMBuffer alloc] initWithPCMFormat:format frameCapacity:clickFrameCount];
    if (![clickFile readIntoBuffer:self.clickBuffer error:&loadError]) {
        if (error) *error = loadError;
        return NO;
    }

    // Load accent sound
    AVAudioFile *accentFile = [[AVAudioFile alloc] initForReading:accentURL error:&loadError];
    if (!accentFile) {
        if (error) *error = loadError;
        return NO;
    }

    AVAudioFrameCount accentFrameCount = (AVAudioFrameCount)accentFile.length;
    self.accentBuffer = [[AVAudioPCMBuffer alloc] initWithPCMFormat:format frameCapacity:accentFrameCount];
    if (![accentFile readIntoBuffer:self.accentBuffer error:&loadError]) {
        if (error) *error = loadError;
        return NO;
    }

    // Connect player node now that we know the format
    [self.engine connect:self.playerNode to:self.engine.mainMixerNode format:format];

    return YES;
}

- (void)setOnBeatCallback:(void (^)(NSInteger, BOOL))callback {
    self.onBeatCallback = callback;
}

- (void)setBpm:(double)bpm {
    _bpm = bpm;
    // Recalculate beat interval in samples
    double sampleRate = self.engine.mainMixerNode.outputFormat.sampleRate;
    if (sampleRate > 0) {
        _beatInterval = (NSInteger)((60.0 / bpm) * sampleRate);
    }
}

// Placeholder start/stop - will be completed in Task 3
- (BOOL)start:(NSError **)error {
    // Will be implemented in Task 3
    return NO;
}

- (void)stop {
    // Will be implemented in Task 3
}

@end
```

```objc
// MARK: - Scheduling Methods (CRITICAL - follow these rules)

- (void)scheduleNextBeats {
    // Calculate current render time
    AVAudioTime *lastRenderTime = self.playerNode.lastRenderTime;
    if (!lastRenderTime || !lastRenderTime.isSampleTimeValid) return;

    AVAudioFramePosition currentSampleTime = lastRenderTime.sampleTime;
    double sampleRate = self.engine.mainMixerNode.outputFormat.sampleRate;

    // Keep scheduling until we have 2-3 beats in the buffer
    while (self.nextBeatSampleTime < currentSampleTime + (self.beatInterval * 3)) {
        BOOL isAccent = (self.currentBeat == 0);
        AVAudioPCMBuffer *buffer = isAccent ? self.accentBuffer : self.clickBuffer;

        AVAudioTime *beatTime = [[AVAudioTime alloc] initWithSampleTime:self.nextBeatSampleTime
                                                               atRate:sampleRate];

        // Schedule the buffer at the precise sample time
        [self.playerNode scheduleBuffer:buffer atTime:beatTime options:0 completionHandler:nil];

        // Fire callback on main thread (NOT in audio path)
        NSInteger beatNum = self.currentBeat;
        dispatch_async(dispatch_get_main_queue(), ^{
            if (self.onBeatCallback) {
                self.onBeatCallback(beatNum, isAccent);
            }
        });

        self.nextBeatSampleTime += self.beatInterval;
        self.currentBeat = (self.currentBeat + 1) % self.beatsPerBar;
    }
}

- (BOOL)start:(NSError **)error {
    if (self.isPlaying) return YES;

    if (!self.clickBuffer || !self.accentBuffer) {
        if (error) {
            *error = [NSError errorWithDomain:@"AudioEngine" code:1
                      userInfo:@{NSLocalizedDescriptionKey: @"Sounds not loaded"}];
        }
        return NO;
    }

    NSError *startError = nil;
    [self.engine prepare];
    if (![self.engine startAndReturnError:&startError]) {
        if (error) *error = startError;
        return NO;
    }

    [self.playerNode play];

    // Calculate initial beat interval
    double sampleRate = self.engine.mainMixerNode.outputFormat.sampleRate;
    self.beatInterval = (NSInteger)((60.0 / self.bpm) * sampleRate);

    // Get current time and start scheduling from next beat boundary
    AVAudioTime *now = self.playerNode.lastRenderTime;
    self.nextBeatSampleTime = now ? now.sampleTime : 0;
    self.currentBeat = 0;

    // Schedule first beats immediately
    [self scheduleNextBeats];

    // Start scheduler timer - fires 2x per beat interval for safety
    NSTimeInterval timerInterval = (60.0 / self.bpm) / 2.0;

    dispatch_queue_t queue = dispatch_get_global_queue(DISPATCH_QUEUE_PRIORITY_HIGH, 0);
    self.schedulerTimer = dispatch_source_create(DISPATCH_SOURCE_TYPE_TIMER, 0, 0, queue);

    dispatch_source_set_timer(self.schedulerTimer,
                              dispatch_time(DISPATCH_TIME_NOW, (int64_t)(timerInterval * NSEC_PER_SEC)),
                              (uint64_t)(timerInterval * NSEC_PER_SEC),
                              (uint64_t)(0.001 * NSEC_PER_SEC));  // 1ms leeway

    __weak typeof(self) weakSelf = self;
    dispatch_source_set_event_handler(self.schedulerTimer, ^{
        [weakSelf scheduleNextBeats];
    });

    dispatch_resume(self.schedulerTimer);

    self.isPlaying = YES;
    return YES;
}

- (void)stop {
    if (!self.isPlaying) return;

    // Cancel timer first
    if (self.schedulerTimer) {
        dispatch_source_cancel(self.schedulerTimer);
        self.schedulerTimer = nil;
    }

    // Stop playback
    [self.playerNode stop];
    [self.engine stop];

    // Reset state
    self.nextBeatSampleTime = 0;
    self.currentBeat = 0;
    self.isPlaying = NO;
}
```

CRITICAL patterns enforced:
1. NO locks in scheduleNextBeats (atomic operations only)
2. NO memory allocation in scheduling path (buffers pre-loaded)
3. NO Swift/ObjC runtime calls in audio path
4. Timer fires on high-priority queue (not main thread)
5. Callbacks dispatched to main thread asynchronously
6. Pre-schedules 2-3 beats ahead using sample-accurate timing
7. Uses lastRenderTime as master clock (not system time)
